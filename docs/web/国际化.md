## element 组件国际化配置

::: tip
全局引入挂载 i18n 方法
:::

```ts
//main.ts
import { i18n } from '/@/i18n/index';
app.use(pinia).use(router)use(i18n).mount('#app');
```

::: tip
i18n 生成方法
:::

```ts
//i18n/index.ts
//引入element几种语言
import enLocale from 'element-plus/es/locale/lang/en';
import zhcnLocale from 'element-plus/es/locale/lang/zh-cn';
import zhtwLocale from 'element-plus/es/locale/lang/zh-tw';

//Define variable content
const messages = {};
const element = { en: enLocale, 'zh-cn': zhcnLocale, 'zh-tw': zhtwLocale };
const itemize = { en: [], 'zh-cn': [], 'zh-tw': [] };
const modules: Record<string, any> = import.meta.glob('./**/*.ts', { eager: true });// 自定义的语言包
![cb5ef790cbbfcab5824131deb23db56](https://blog.babade.asia/I18n/cb5ef790cbbfcab5824131deb23db56.png)
// https://vitejs.cn/vite3-cn/guide/features.html#glob-import
for (const path in modules) {
	const key = path.match(/(\S+)\/(\S+).ts/);
	if (itemize[key![2]]) itemize[key![2]].push(modules[path].default);
	else itemize[key![2]] = modules[path];
}

function mergeArrObj<T>(list: T, key: string) {
	let obj = {};
	list[key].forEach((i: EmptyObjectType) => {
		obj = Object.assign({}, obj, i);
	});
	return obj;
}

for (const key in itemize) {
	messages[key] = {
		name: key,
		el: element[key].el,
		message: mergeArrObj(itemize, key),
	};
}
//注意 message el部分是element国际化内容， message部分是自定义的国际化内容

const stores = useThemeConfig(pinia);
const { themeConfig } = storeToRefs(stores);

// https://vue-i18n.intlify.dev/guide/essentials/fallback.html#explicit-fallback-with-one-locale
export const i18n = createI18n({
	legacy: false, //vue3 composition Api这里必须要用false
	silentTranslationWarn: true,
	missingWarn: false,
	silentFallbackWarn: true,
	fallbackWarn: false,
	locale: themeConfig.value.globalI18n,
	fallbackLocale: zhcnLocale.name,
	messages,
});
```

::: tip
页面中使用
:::

```ts
// 58行 自定义了页面国际化内容 message参数
<p>{{ $t('message.layout.sixDefaults') }}</p>
//App.vue
import { useI18n } from "vue-i18n";
const { messages, locale } = useI18n();

// messages 是包含了element和自定义国际化内容的部分，locale是当前语言
// 切换语言时，改变locale值即可生效
const onLanguageChange = (lang: string) => {
  locale.value = lang;
};


//element组件locale属性控制element国际化内容，locale发生改变，message去寻找对应的语言包
<el-config-provider :size="getGlobalComponentSize" :locale="getGlobalI18n">
    //内容
</el-config-provider>

const getGlobalI18n = computed(() => {
  return messages.value[locale.value];
});



//源码是
import ElementPlus from 'element-plus'
import zhCn from 'element-plus/es/locale/lang/zh-cn'

<template>
  <el-config-provider :locale="zhCn">
    <app />
  </el-config-provider>
</template>
import zhCn from 'element-plus/es/locale/lang/zh-cn'
//zhCn的结构如下
declare const _default: {
    name: string;
    el: {
        breadcrumb: {
            label: string;
        };
        .......
    };
};
```
